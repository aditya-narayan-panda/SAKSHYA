# --- Cryptography ---

# pins 50.0.0. With an older version the backend REFUSES TO BOOT (no silent classical fallback);
# the only escape hatch is the developer-only flag SAKSHYA_ALLOW_FALLBACK=true, never honoured
# when SAKSHYA_APP_ENV=production. GET /api/system/crypto-status reports what is really active.
cryptography==50.0.0

# --- Document / watermark / report generation (all local, no cloud) ---

reportlab==5.0.1

# --- HTTP client (test client + reserved for future inter-node calls) ---
